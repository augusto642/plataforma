def two_sum_optimal(nums: list[int], target: int) -> list[int]:
"""
Encontra os índices de dois números que somam o target.
Complexidade: O(N) tempo e O(N) espaço residente.
"""
seen = {}
for i, num in enumerate(nums):
diff = target - num
if diff in seen:
return [seen[diff], i]
seen[num] = i
return []
# Caso de teste com 10.000 elementos
import time
start = time.perf_counter()
res = two_sum_optimal(list(range(10000)), 19997)
elapsed = (time.perf_counter() - start) * 1000
print(f"Resultado: {res} | Tempo: {elapsed:.3f}ms")
